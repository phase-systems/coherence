cmake_minimum_required(VERSION 3.22)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "")
project(PlugHost VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../deps/JUCE" juce)
juce_add_console_app(PlugHost PRODUCT_NAME "PlugHost")
target_sources(PlugHost PRIVATE PlugHost.cpp)
target_compile_definitions(PlugHost PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(PlugHost PRIVATE juce::juce_audio_processors juce::juce_audio_formats juce::juce_recommended_config_flags "-framework AudioUnit" "-framework CoreAudioKit")
